        if(arr[arr.length-1]<n) return arr.length;
        int left=0;
        int right=arr.length-1;
        while(left<right){
          int  m=left+(right-left)/2;
            if(arr[m]==n){
                return m;
            }
            else if(arr[m]<n){
                left=m+1;
            }
            else{
                right=m-1;
            }
        }
        return left;
        
    }
}
